class User {
  - userId: int
  - username: string
  - password: string
  - email: string
  - role: string
  - registrationDate: datetime
  + login(): void
  + logout(): void
  + updateProfile(): void
}

class Admin {
  - adminLevel: string
  + addBook(): void
  + updateBook(): void
  + deleteBook(): void
  + viewAllBookings(): void
  + approveBooking(): void
  + declineBooking(): void
  + viewAllClients(): void
  + manageBookStatus(): void
}

class Client {
  - phoneNumber: string
  - address: string
  + browseBooks(): void
  + searchBooks(): void
  + bookABook(): void
  + viewMyBookings(): void
  + trackBookingStatus(): void
}

class Book {
  - bookId: int
  - title: string
  - author: string
  - category: string
  - ISBN: string
  - publicationYear: int
  - description: string
  - status: string
  - availableCopies: int

  + updateStatus(): void
  + checkAvailability(): boolean
}

class Booking {
  - bookingId: int
  - userId: int
  - bookId: int
  - bookingDate: datetime
  - dueDate: datetime
  - returnDate: datetime
  - status: string
  + createBooking(): void
  + updateStatus(): void
  + cancelBooking(): void

}

class Category {
  - categoryId: int
  - categoryName: string
  - description: string
  + addCategory(): void
  + updateCategory(): void
}
